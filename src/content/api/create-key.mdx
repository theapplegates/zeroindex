      -d '{ "name": "Storefront search", "kind": "search", "indexes": ["products"] }'

    const key = await client.keys.create({

    key = client.keys.create(name="Storefront search", kind="search", indexes=["products"])

Requires an admin key.

<Callout type="warning" title="Save the key now">
  The response is the only time you see the full key. Store it in your secrets manager before you close the page.
</Callout>

## Body

<ParamField name="name" type="string" required>
  A label to recognise the key in the dashboard, like the app that uses it.
</ParamField>
<ParamField name="kind" type="admin | search" required>
  Admin keys can call every endpoint. Search keys can only search.
</ParamField>
<ParamField name="indexes" type="string[]" default='["*"]'>
  The indexes the key can reach.
</ParamField>
